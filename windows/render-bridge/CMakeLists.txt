cmake_minimum_required(VERSION 3.24)
project(RtxForgeRenderBridge LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(render_bridge_model_tests tests/model_tests.cpp)
target_include_directories(render_bridge_model_tests PRIVATE include)
add_test(NAME resource_and_descriptor_lifetimes COMMAND render_bridge_model_tests)
if(WIN32)
  set(RESHADE_SDK_DIR "" CACHE PATH "Pinned ReShade include directory")
  if(NOT EXISTS "${RESHADE_SDK_DIR}/reshade.hpp")
    message(FATAL_ERROR "Set RESHADE_SDK_DIR to ReShade 6.8.0 include/; see README.md")
  endif()
  add_library(render_bridge SHARED src/addon.cpp)
  target_include_directories(render_bridge PRIVATE include "${RESHADE_SDK_DIR}")
  target_compile_definitions(render_bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(render_bridge PRIVATE bcrypt user32)
  set_target_properties(render_bridge PROPERTIES PREFIX "" OUTPUT_NAME "rtxforge-render-bridge" SUFFIX ".addon64")
  add_executable(render_bridge_harness tests/d3d12_harness.cpp)
  target_compile_definitions(render_bridge_harness PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(render_bridge_harness PRIVATE d3d12 dxgi d3dcompiler user32)
  add_library(render_bridge_registration_probe SHARED tests/registration_probe.cpp)
  target_include_directories(render_bridge_registration_probe PRIVATE "${RESHADE_SDK_DIR}")
  set_target_properties(render_bridge_registration_probe PROPERTIES PREFIX "" SUFFIX ".addon64")
  if(MSVC)
    target_compile_options(render_bridge PRIVATE /W4 /EHsc /permissive- /Zi)
    target_link_options(render_bridge PRIVATE /DEBUG:FULL)
    target_compile_options(render_bridge_harness PRIVATE /Zi)
    target_link_options(render_bridge_harness PRIVATE /DEBUG:FULL)
    target_compile_options(render_bridge_model_tests PRIVATE /W4 /EHsc /permissive-)
  endif()
endif()
