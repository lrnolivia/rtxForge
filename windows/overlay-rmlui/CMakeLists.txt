cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(ForgeGlassOverlay LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT RMLUI_SOURCE OR NOT FREETYPE_ROOT)
  message(FATAL_ERROR "Windows, pinned RMLUI_SOURCE and FREETYPE_ROOT are required")
endif()
add_library(Freetype::Freetype STATIC IMPORTED GLOBAL)
set_target_properties(Freetype::Freetype PROPERTIES IMPORTED_LOCATION "${FREETYPE_ROOT}/freetype.lib" INTERFACE_INCLUDE_DIRECTORIES "${FREETYPE_ROOT}")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
set(RMLUI_TESTS OFF CACHE BOOL "" FORCE)
set(RMLUI_LUA_BINDINGS OFF CACHE BOOL "" FORCE)
set(RMLUI_SHELL OFF CACHE BOOL "" FORCE)
add_subdirectory("${RMLUI_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/rmlui")
find_program(GLASS_POWERSHELL NAMES pwsh powershell REQUIRED)
set(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(GLOB GLASS_ICONS "${REPO}/prototypes/overlay-rmlui/icons/*-light.png")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/GlassAssets.rc" "${CMAKE_CURRENT_BINARY_DIR}/GlassAssets.generated.h"
  COMMAND "${GLASS_POWERSHELL}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/Embed-Assets.ps1" -Output "${CMAKE_CURRENT_BINARY_DIR}"
  DEPENDS Embed-Assets.ps1 "${REPO}/prototypes/overlay-rmlui/overlay.rml" "${REPO}/prototypes/overlay-rmlui/overlay.rcss" "${REPO}/prototypes/overlay-rmlui/logo-mark.png" ${GLASS_ICONS})
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/PrototypeControls.generated.h"
  COMMAND "${GLASS_POWERSHELL}" -NoProfile -File "${REPO}/prototypes/overlay-rmlui/Generate-Controls.ps1" -Output "${CMAKE_CURRENT_BINARY_DIR}/PrototypeControls.generated.h"
  DEPENDS "${REPO}/prototypes/overlay-rmlui/Generate-Controls.ps1" "${REPO}/prototypes/overlay-rmlui/scenes.json" "${REPO}/providers/skynizz-bridge/controls.json" "${REPO}/providers/renodx-hdr-settings.json")
add_library(rtxforge-glass SHARED GlassRuntime.cpp GlassDx12.cpp "${RMLUI_SOURCE}/Backends/RmlUi_Renderer_DX11.cpp" "${CMAKE_CURRENT_BINARY_DIR}/GlassAssets.rc" "${CMAKE_CURRENT_BINARY_DIR}/GlassAssets.generated.h" "${CMAKE_CURRENT_BINARY_DIR}/PrototypeControls.generated.h")
target_compile_features(rtxforge-glass PRIVATE cxx_std_20)
target_compile_options(rtxforge-glass PRIVATE /utf-8)
target_compile_definitions(rtxforge-glass PRIVATE NOMINMAX UNICODE _UNICODE NUM_MSAA_SAMPLES=4)
target_include_directories(rtxforge-glass PRIVATE "${RMLUI_SOURCE}/Backends" "${REPO}/providers/skynizz-bridge" "${REPO}/providers/glass-overlay" "${CMAKE_CURRENT_BINARY_DIR}" "${FREETYPE_ROOT}/../nlohmann")
target_link_libraries(rtxforge-glass PRIVATE RmlUi::Core d3d11 d3d12 dxgi d3dcompiler windowscodecs ole32 user32)
add_executable(GlassSmokeTest SmokeTest.cpp)
target_compile_features(GlassSmokeTest PRIVATE cxx_std_20)
target_compile_definitions(GlassSmokeTest PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_options(GlassSmokeTest PRIVATE /utf-8)
target_include_directories(GlassSmokeTest PRIVATE "${REPO}/providers/glass-overlay")
target_link_libraries(GlassSmokeTest PRIVATE d3d11 d3d12 dxgi)
add_executable(OverlayPreview "${REPO}/prototypes/overlay-rmlui/Prototype.cpp" "${RMLUI_SOURCE}/Backends/RmlUi_Renderer_DX11.cpp" "${CMAKE_CURRENT_BINARY_DIR}/PrototypeControls.generated.h")
target_compile_features(OverlayPreview PRIVATE cxx_std_20)
target_compile_options(OverlayPreview PRIVATE /utf-8)
target_compile_definitions(OverlayPreview PRIVATE NOMINMAX UNICODE _UNICODE NUM_MSAA_SAMPLES=4)
target_include_directories(OverlayPreview PRIVATE "${RMLUI_SOURCE}/Backends" "${REPO}/providers/skynizz-bridge" "${CMAKE_CURRENT_BINARY_DIR}" "${FREETYPE_ROOT}/../nlohmann")
target_link_libraries(OverlayPreview PRIVATE RmlUi::Core d3d11 dxgi d3dcompiler windowscodecs ole32 user32)
