cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(OverlayRmlUiPrototype LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "This prototype evaluates the existing Windows DirectX 11 overlay boundary.")
endif()
if(NOT RMLUI_SOURCE OR NOT FREETYPE_ROOT)
  message(FATAL_ERROR "Supply the pinned RMLUI_SOURCE and the existing FREETYPE_ROOT.")
endif()
add_library(Freetype::Freetype STATIC IMPORTED GLOBAL)
set_target_properties(Freetype::Freetype PROPERTIES IMPORTED_LOCATION "${FREETYPE_ROOT}/freetype.lib" INTERFACE_INCLUDE_DIRECTORIES "${FREETYPE_ROOT}")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
set(RMLUI_TESTS OFF CACHE BOOL "" FORCE)
set(RMLUI_LUA_BINDINGS OFF CACHE BOOL "" FORCE)
set(RMLUI_SHELL OFF CACHE BOOL "" FORCE)
set(RMLUI_PRECOMPILED_HEADERS ON CACHE BOOL "" FORCE)
add_subdirectory("${RMLUI_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/rmlui")
add_executable(OverlayPrototype Prototype.cpp "${RMLUI_SOURCE}/Backends/RmlUi_Renderer_DX11.cpp")
find_program(PROTOTYPE_POWERSHELL NAMES pwsh powershell REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/PrototypeControls.generated.h"
  COMMAND "${PROTOTYPE_POWERSHELL}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/Generate-Controls.ps1" -Output "${CMAKE_CURRENT_BINARY_DIR}/PrototypeControls.generated.h"
  DEPENDS Generate-Controls.ps1 scenes.json "${CMAKE_CURRENT_SOURCE_DIR}/../../providers/skynizz-bridge/controls.json")
target_sources(OverlayPrototype PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/PrototypeControls.generated.h")
target_compile_features(OverlayPrototype PRIVATE cxx_std_20)
target_compile_options(OverlayPrototype PRIVATE /utf-8)
target_compile_definitions(OverlayPrototype PRIVATE NOMINMAX UNICODE _UNICODE NUM_MSAA_SAMPLES=4)
target_include_directories(OverlayPrototype PRIVATE "${RMLUI_SOURCE}/Backends" "${CMAKE_CURRENT_SOURCE_DIR}/../../providers/skynizz-bridge" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(OverlayPrototype PRIVATE RmlUi::Core d3d11 dxgi d3dcompiler windowscodecs ole32 user32)
